REPLAY CLOCK (owns historical time)
for replay_date in experiment.business_dates:
    context = build_overall_default_context(
        point_in_time_data(replay_date),
        portfolio_after_approved_events,
        metric_pack.calculate()
    )

    # WORKFLOW CYCLES share the same frozen context and date
    for workflow in enabled_workflows:
        adapter  = runtime_adapters[workflow.runtime]
        topology = topology_factory[workflow.topology]
        output   = adapter.run(topology, context, workflow.capabilities)
        decision = deterministic_critic.validate(output, context)
        audit_log.append(decision)

        if decision.requires_attention:
            pause_replay_at(decision)       # human accepts or rejects
        else:
            decision.review = "AUTO_CLEARED"
